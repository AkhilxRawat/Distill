services:
  tempo:
    image: grafana/tempo:2.6.1
    command: ["-config.file=/etc/tempo.yaml"]
    networks:
      - internal
    volumes:
      - ./tempo.yaml:/etc/tempo.yaml

  prometheus:
    image: prom/prometheus:latest
    command:
      - "--config.file=/etc/prometheus/prometheus.yml"
      - "--enable-feature=exemplar-storage"
      - "--web.enable-remote-write-receiver"
    networks:
      - internal
    volumes:
      - ./prometheus.yml:/etc/prometheus/prometheus.yml

  grafana:
    image: grafana/grafana:latest
    networks:
      - public
      - internal
    ports:
      - "127.0.0.1:3001:3000"
    environment:
      - GF_AUTH_ANONYMOUS_ENABLED=true
      - GF_AUTH_ANONYMOUS_ORG_ROLE=Admin
      - GF_FEATURE_TOGGLES_ENABLE=traceqlEditor
    volumes:
      - ./grafana-datasources.yml:/etc/grafana/provisioning/datasources/datasources.yml
    depends_on:
      - prometheus
      - tempo

networks:
  public:
  internal:













services:
  mongodb:
    image: mongo:7
    networks:
      - internal
    volumes:
      - mongo_data:/data/db

  ingestion:
    build:
      context: .
      dockerfile: services/ingestion/Dockerfile
    networks:
      - internal
    environment:
      - STORAGE_HOST=storage
      - STORAGE_PORT=50053
      - PROCESSING_HOST=processing
      - PROCESSING_PORT=50052
      - OTEL_EXPORTER_OTLP_ENDPOINT=http://tempo:4318
    depends_on:
      - mongodb
      - storage
      - processing

  processing:
    build:
      context: .
      dockerfile: services/processing/Dockerfile
    networks:
      - internal
    environment:
      - STORAGE_HOST=storage
      - STORAGE_PORT=50053
      - GEMINI_API_KEY=${GEMINI_API_KEY}
      - OTEL_EXPORTER_OTLP_ENDPOINT=http://tempo:4318
    depends_on:
      - mongodb
      - storage

  storage:
    build:
      context: .
      dockerfile: services/storage/Dockerfile
    networks:
      - internal
    environment:
      - MONGODB_URI=mongodb://mongodb:27017/distill
      - OTEL_EXPORTER_OTLP_ENDPOINT=http://tempo:4318
    depends_on:
      - mongodb

  gateway:  
    build:
      context: .
      dockerfile: services/gateway/Dockerfile
    networks:
      - public
      - internal
    ports:
      - "127.0.0.1:3000:3000"
    environment:
      - INGESTION_HOST=ingestion
      - INGESTION_PORT=50051
      - STORAGE_HOST=storage
      - STORAGE_PORT=50053
      - JWT_SECRET=${JWT_SECRET}
      - OTEL_EXPORTER_OTLP_ENDPOINT=http://tempo:4318
    depends_on:
      - ingestion
      - storage

  frontend:
    build:
      context: .
      dockerfile: frontend/Dockerfile
      args:
        NEXT_PUBLIC_API_URL: http://localhost:3000
    networks:
      - public
    ports:
      - "127.0.0.1:4000:3000"
    environment:
      - NEXT_PUBLIC_API_URL=http://localhost:3000
    depends_on:
      - gateway
